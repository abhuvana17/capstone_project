SQL:


SELECT
    c.category_name,

    COUNT(b.book_id)
        AS book_count,

    ROUND(
        AVG(b.price_gbp),
        2
    ) AS average_price_gbp

FROM categories c

JOIN books b
ON c.category_id =
   b.category_id

GROUP BY
    c.category_name

ORDER BY
    book_count DESC;

        Philosophy          11              33.56

================================================================================

SQL:


SELECT
    c.category_name,

    COUNT(b.book_id)
        AS book_count

FROM categories c

JOIN books b
ON c.category_id =
   b.category_id

GROUP BY
    c.category_name

HAVING
    COUNT(b.book_id) >= 5

ORDER BY
    book_count DESC;



OUTPUT:
     category_name  book_count
    Sequential Art          75
           Mystery          32
Historical Fiction          26
          Classics          19
            Travel          11
        Philosophy          11

QUERY 7
================================================================================

SQL:


SELECT

    COUNT(*) AS total_books,

    ROUND(
        MIN(price_gbp), 2
    ) AS minimum_price_gbp,

    ROUND(
        MAX(price_gbp), 2
    ) AS maximum_price_gbp,

    ROUND(
        AVG(price_gbp), 2
    ) AS average_price_gbp,

    SUM(
        CASE
            WHEN in_stock = 1
            THEN 1
            ELSE 0
        END
    ) AS books_in_stock

FROM books;



OUTPUT:

SQL:


SELECT

    title,
    rating,

    CASE

        WHEN rating = 5
            THEN 'Excellent'

        WHEN rating = 4
            THEN 'Very Good'

        WHEN rating = 3
            THEN 'Good'

        ELSE 'Below Average'

    END AS rating_label

FROM books

ORDER BY rating DESC

LIMIT 15;
